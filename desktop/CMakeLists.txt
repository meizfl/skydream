cmake_minimum_required(VERSION 3.18)
project(Skydream LANGUAGES C CXX)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/f3daudio/LICENSE"
        DESTINATION "share/licenses/skydream" RENAME "F3DAudio-LICENSE")

if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Skydream requires a 64-bit host")
endif()
if(MSVC)
    message(FATAL_ERROR "Generated code requires GCC/Clang extensions. On Windows use MinGW-w64 (MSYS2 UCRT64).")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS ON)
set(CMAKE_CXX_STANDARD 17)
include(GNUInstallDirs)
find_package(SDL2 REQUIRED CONFIG)
# SDL loads the Vulkan loader on all platforms; only headers are linked.
find_path(SKYDREAM_VULKAN_INCLUDE vulkan/vulkan.h HINTS "$ENV{VULKAN_SDK}/include" "$ENV{VULKAN_SDK}/Include" REQUIRED)
if(NOT TARGET Vulkan::Headers)
    add_library(Vulkan::Headers INTERFACE IMPORTED)
    set_target_properties(Vulkan::Headers PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${SKYDREAM_VULKAN_INCLUDE}")
endif()
find_path(SKYDREAM_SDL_PARENT SDL2/SDL.h REQUIRED)
include_directories(SYSTEM "${SKYDREAM_SDL_PARENT}")
find_package(Threads REQUIRED)
option(SKYDREAM_FFMPEG "Enable native video and compressed game audio (required for a complete game build)" ON)
add_library(skydream_media INTERFACE)
if(SKYDREAM_FFMPEG)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavutil libswresample libavformat libswscale)
    target_link_libraries(skydream_media INTERFACE PkgConfig::FFMPEG)
    target_compile_definitions(skydream_media INTERFACE LB_HAVE_FFMPEG)
else()
    message(WARNING "FFmpeg disabled: build check only; Bink videos and compressed voices are unavailable")
endif()
# The shared Android DXVK sources link to these names.
if(NOT TARGET SDL2)
    add_library(SDL2 INTERFACE)
    target_link_libraries(SDL2 INTERFACE SDL2::SDL2)
endif()

option(SKYDREAM_BUNDLE_DEPS "Collect and relocate Windows/macOS runtime dependencies during install" OFF)
option(SKYDREAM_ROOT_PACKAGE "Install Skydream executable beside Data rather than under bin" OFF)
if(SKYDREAM_ROOT_PACKAGE)
    set(SKYDREAM_EXEC_DEST ".")
else()
    set(SKYDREAM_EXEC_DEST "${CMAKE_INSTALL_BINDIR}")
endif()

get_filename_component(PORT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
install(FILES "${PORT}/third_party/stb/LICENSE" DESTINATION "share/licenses/skydream" RENAME "stb-LICENSE")
install(FILES "${PORT}/third_party/vkd3d/LICENSE" DESTINATION "share/licenses/skydream" RENAME "vkd3d-LGPL-2.1.txt")
if(WIN32)
    set(_shader_default "${PORT}/build/native-shader/bin/libvkd3d-shader-1.dll")
elseif(APPLE)
    set(_shader_default "${PORT}/build/native-shader/lib/libvkd3d-shader.1.dylib")
else()
    set(_shader_default "${PORT}/build/native-shader/lib/libvkd3d-shader.so.1")
endif()
set(SKYDREAM_SHADER_LIBRARY "${_shader_default}" CACHE FILEPATH "Native vkd3d-shader 2.0 shared library for the target OS")
if(EXISTS "${SKYDREAM_SHADER_LIBRARY}")
    if(WIN32)
        install(FILES "${SKYDREAM_SHADER_LIBRARY}" DESTINATION "${SKYDREAM_EXEC_DEST}")
    else()
        install(FILES "${SKYDREAM_SHADER_LIBRARY}" DESTINATION "${CMAKE_INSTALL_LIBDIR}/skyrim-native")
    endif()
else()
    message(STATUS "Native HLSL compiler not bundled: provide target-OS vkd3d-shader 2.0 using SKYDREAM_SHADER_LIBRARY")
endif()

option(LB_BUILD_GAME "Build the game (requires generated code from your own game files)" ON)
option(SKYDREAM_SKSE "Enable experimental independent SKSE Papyrus compatibility (not full SKSE)" OFF)
option(SKYDREAM_SKYUI "Enable the experimental SkyUI interface by default" OFF)
if(SKYDREAM_SKYUI AND NOT SKYDREAM_SKSE)
    message(FATAL_ERROR "SKYDREAM_SKYUI requires SKYDREAM_SKSE")
endif()
set(LB_GEN_JOBS 2 CACHE STRING "Maximum simultaneous generated-code compilations")
set_property(GLOBAL PROPERTY JOB_POOLS generated=${LB_GEN_JOBS})
if(LB_BUILD_GAME)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    foreach(module exe)
        if(NOT EXISTS "${PORT}/gen/${module}_table.c")
            message(FATAL_ERROR "Missing gen/${module}_table.c. Run: sh tools/regen_gen.sh GAME_DIR. Use -DLB_BUILD_GAME=OFF to check DXVK and the host runtime only.")
        endif()
    endforeach()
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${PORT}/tools/prepare_skse_hooks.py" "${PORT}/gen"
        RESULT_VARIABLE SKSE_HOOK_RESULT)
    if(NOT SKSE_HOOK_RESULT EQUAL 0)
        message(FATAL_ERROR "Cannot prepare native host hooks in generated game code")
    endif()
endif()

if(EXISTS "${PORT}/third_party/dxvk/CMakeLists.txt")
    set(SKYDREAM_DXVK "${PORT}/third_party/dxvk")
else()
    set(SKYDREAM_DXVK "${PORT}/android/app/jni/dxvk")
endif()
add_subdirectory("${SKYDREAM_DXVK}" dxvk)
if(APPLE)
    set_target_properties(dxvk_d3d9 PROPERTIES INSTALL_RPATH "@loader_path")
elseif(UNIX)
    set_target_properties(dxvk_d3d9 PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
file(GLOB RT_SRC CONFIGURE_DEPENDS "${PORT}/runtime/*.c" "${PORT}/runtime/*.cpp")
list(FILTER RT_SRC EXCLUDE REGEX "/host\\.c$")
add_library(skydream_host STATIC "${PORT}/runtime/host.c")
target_link_libraries(skydream_host PUBLIC SDL2::SDL2)
target_compile_definitions(skydream_host PRIVATE _FILE_OFFSET_BITS=64)
include(CTest)
if(BUILD_TESTING)
    add_executable(skydream_host_smoke "${PORT}/tests/host_smoke.c")
    target_include_directories(skydream_host_smoke PRIVATE "${PORT}/runtime")
    target_link_libraries(skydream_host_smoke PRIVATE skydream_host Threads::Threads)
    target_compile_options(skydream_host_smoke PRIVATE -UNDEBUG)
    if(WIN32 AND TARGET SDL2::SDL2)
        get_target_property(_sdl_type SDL2::SDL2 TYPE)
        if(_sdl_type STREQUAL "SHARED_LIBRARY")
            add_custom_command(TARGET skydream_host_smoke POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:SDL2::SDL2>" "$<TARGET_FILE_DIR:skydream_host_smoke>")
        endif()
    endif()
    add_test(NAME host-platform COMMAND skydream_host_smoke)
endif()
add_library(lb_runtime OBJECT ${RT_SRC} "${PORT}/third_party/f3daudio/F3DAudio.c")
target_include_directories(lb_runtime PRIVATE "${PORT}/runtime" "${PORT}/gen" "${PORT}/third_party/vkd3d/include")
target_compile_definitions(lb_runtime PRIVATE GAME_SKYRIM DXVK_NATIVE _FILE_OFFSET_BITS=64 SDL_MAIN_HANDLED)
if(SKYDREAM_SKSE)
    target_compile_definitions(lb_runtime PRIVATE SKYDREAM_SKSE=1)
    if(SKYDREAM_SKYUI)
        target_compile_definitions(lb_runtime PRIVATE SKYDREAM_SKYUI=1)
    endif()
    message(STATUS "SKSE compatibility: experimental Papyrus subset; no Windows DLL plugins")
else()
    list(FILTER RT_SRC EXCLUDE REGEX "skse_compat_values\\.c$")
    set_property(TARGET lb_runtime PROPERTY SOURCES ${RT_SRC})
endif()
target_compile_options(lb_runtime PRIVATE -O2 -fno-strict-aliasing -fwrapv -fsigned-char)
target_link_libraries(lb_runtime PRIVATE SDL2::SDL2 dxvk_d3d9 Threads::Threads skydream_media skydream_host ${CMAKE_DL_LIBS})

if(LB_BUILD_GAME)
    file(GLOB GEN_SRC CONFIGURE_DEPENDS "${PORT}/gen/*.c")
    list(FILTER GEN_SRC EXCLUDE REGEX "/(x3da|bink|d3dx9)_[^/]*\\.c$")
    add_library(lb_generated OBJECT ${GEN_SRC})
    set_property(TARGET lb_generated PROPERTY JOB_POOL_COMPILE generated)
    target_include_directories(lb_generated PRIVATE "${PORT}/runtime" "${PORT}/gen")
    target_compile_definitions(lb_generated PRIVATE GAME_SKYRIM)
    target_compile_options(lb_generated PRIVATE -O2 -g0 -fno-strict-aliasing -fwrapv -fsigned-char -w)
    option(LB_STRICT_ORDER "Full x86 memory ordering on ARM64" OFF)
    if(LB_STRICT_ORDER)
        target_compile_definitions(lb_generated PRIVATE LB_STRICT_ORDER)
    endif()
    set(SKYDREAM_GAME_IMAGE "" CACHE FILEPATH "Unpacked TESV used only to prepare link-time game data")
    if(SKYDREAM_GAME_IMAGE)
        execute_process(COMMAND "${Python3_EXECUTABLE}" "${PORT}/tools/embed_game_image.py"
            "${SKYDREAM_GAME_IMAGE}" "${PORT}/gen/image" RESULT_VARIABLE IMAGE_RESULT)
        if(NOT IMAGE_RESULT EQUAL 0)
            message(FATAL_ERROR "Cannot prepare embedded game data")
        endif()
    endif()
    if(NOT EXISTS "${PORT}/gen/image/skyrim_image.c" OR NOT EXISTS "${PORT}/gen/image/skyrim_image.bin")
        message(FATAL_ERROR "Missing link-time game data. Run tools/embed_game_image.py UNPACKED_TESV gen/image, or regenerate game code.")
    endif()
    enable_language(ASM)
    set(SKYDREAM_IMAGE_BLOB "${PORT}/gen/image/skyrim_image.bin")
    configure_file("${PORT}/runtime/embedded_image.S.in" "${CMAKE_CURRENT_BINARY_DIR}/skyrim_image.S" @ONLY)
    set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/skyrim_image.S" PROPERTIES OBJECT_DEPENDS "${SKYDREAM_IMAGE_BLOB}")
    add_library(lb_image OBJECT "${PORT}/gen/image/skyrim_image.c" "${CMAKE_CURRENT_BINARY_DIR}/skyrim_image.S")
    target_include_directories(lb_image PRIVATE "${PORT}/runtime")
    target_compile_definitions(lb_runtime PRIVATE SKYDREAM_EMBEDDED_IMAGE=1)
    add_executable(lb $<TARGET_OBJECTS:lb_runtime> $<TARGET_OBJECTS:lb_generated> $<TARGET_OBJECTS:lb_image>)
    if(SKYDREAM_ROOT_PACKAGE)
        set_target_properties(lb PROPERTIES OUTPUT_NAME Skydream)
    endif()
    set_target_properties(lb PROPERTIES LINKER_LANGUAGE CXX
        RUNTIME_OUTPUT_DIRECTORY "${PORT}/build"
        BUILD_RPATH "${PORT}/build/native-shader/lib"
        INSTALL_RPATH "$ORIGIN/lib/skyrim-native;$ORIGIN/../${CMAKE_INSTALL_LIBDIR}/skyrim-native")
    if(UNIX AND NOT APPLE)
        target_link_options(lb PRIVATE -rdynamic)
    endif()
    if(APPLE)
        set_target_properties(lb PROPERTIES INSTALL_RPATH "@loader_path/lib/skyrim-native;@loader_path/../${CMAKE_INSTALL_LIBDIR}/skyrim-native")
    endif()
    target_link_libraries(lb PRIVATE SDL2::SDL2 dxvk_d3d9 Threads::Threads skydream_media skydream_host ${CMAKE_DL_LIBS})
    if(WIN32)
        add_custom_command(TARGET lb POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:dxvk_d3d9>" "$<TARGET_FILE_DIR:lb>")
    endif()
    if(BUILD_TESTING)
        add_test(NAME embedded-game COMMAND lb --embedded-smoke)
        set_tests_properties(embedded-game PROPERTIES TIMEOUT 60)
    endif()
    if(SKYDREAM_ROOT_PACKAGE)
        install(FILES "${PORT}/docs/ROOT_PACKAGE.txt" DESTINATION "." RENAME README.txt)
        if(EXISTS "${PORT}/LICENSE")
            install(FILES "${PORT}/LICENSE" DESTINATION "share/licenses/skydream")
        endif()
    endif()
    install(TARGETS lb RUNTIME DESTINATION "${SKYDREAM_EXEC_DEST}")
    install(TARGETS dxvk_d3d9 LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/skyrim-native RUNTIME DESTINATION "${SKYDREAM_EXEC_DEST}")
    if(SKYDREAM_BUNDLE_DEPS)
        if(CMAKE_CROSSCOMPILING OR (NOT WIN32 AND NOT APPLE) OR NOT SKYDREAM_ROOT_PACKAGE OR NOT SKYDREAM_FFMPEG)
            message(FATAL_ERROR "Dependency bundling requires a native Windows/macOS full root package with FFmpeg")
        endif()
        if(NOT EXISTS "${SKYDREAM_SHADER_LIBRARY}")
            message(FATAL_ERROR "Build the target native shader compiler first: sh tools/build-desktop-shader.sh")
        endif()
        install(CODE "
            execute_process(COMMAND \"${Python3_EXECUTABLE}\" \"${PORT}/tools/bundle_desktop.py\" \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}\"
                WORKING_DIRECTORY \"${PORT}\" RESULT_VARIABLE bundle_result)
            if(NOT bundle_result EQUAL 0)
                message(FATAL_ERROR \"Cannot bundle desktop runtime dependencies\")
            endif()
        ")
    endif()
endif()
